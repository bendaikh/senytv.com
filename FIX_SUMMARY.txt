Before:

      ['email' => $request->email],
      [
          'name' => $request->first_name . ' ' . $request->last_name,
          'password' => bcrypt(uniqid()),
      ]
  );

After:
  $user = User::firstOrCreate(
      ['email' => $request->email],
      [
          'full_name' => $request->first_name . ' ' . $request->last_name,
          'phone_number' => $request->phone,
          'country' => $request->country,
      ]
  );
